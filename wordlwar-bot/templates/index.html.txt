    <script>
        const tg = window.Telegram.WebApp;
        tg.expand();
        
        const user = tg.initDataUnsafe?.user;
        const userId = user?.id || 12345678;
        const userName = user?.first_name || "ایران";

        async function loadData() {
            const res = await fetch(`/api/get_country/${userId}`);
            if (res.status === 404) {
                await fetch('/api/login', {
                    method: 'POST',
                    headers: {'Content-Type': 'application/json'},
                    body: JSON.stringify({ user_id: userId, country_name: userName, superpower_key: "Iran" })
                });
                return loadData();
            }
            const data = await res.json();
            
            document.getElementById('c-name').innerText = data.country_name;
            document.getElementById('c-pack').innerText = data.special_pack;
            document.getElementById('c-cash').innerText = data.cash.toLocaleString();
            document.getElementById('c-oil').innerText = data.oil.toLocaleString();
            document.getElementById('c-gold').innerText = data.gold.toLocaleString();
            document.getElementById('c-iron').innerText = data.iron.toLocaleString();
            document.getElementById('c-uranium').innerText = data.uranium.toLocaleString();

            document.getElementById('cnt-oil').innerText = data.buildings.oil_mines;
            document.getElementById('cnt-gold').innerText = data.buildings.gold_mines;
            document.getElementById('cnt-iron').innerText = data.buildings.iron_mines;
            document.getElementById('cnt-uranium').innerText = data.buildings.uranium_mines;
            document.getElementById('cnt-cash').innerText = data.buildings.cash_factories;
        }

        async function build(type) {
            const res = await fetch('/api/build', {
                method: 'POST',
                headers: {'Content-Type': 'application/json'},
                body: JSON.stringify({ user_id: userId, type: type })
            });
            const result = await res.json();
            if (res.ok) {
                loadData();
            } else {
                alert(result.error);
            }
        }

        loadData();
    </script>
